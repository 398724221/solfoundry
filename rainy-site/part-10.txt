      function voiceTo(targetId, button){
        const SR=window.SpeechRecognition||window.webkitSpeechRecognition;
        if(!SR){toast('当前浏览器不支持语音识别，可继续使用系统语音输入。');return;}
        const rec=new SR(); rec.lang='zh-CN';rec.continuous=true;rec.interimResults=true;
        const original=button.textContent; button.textContent='正在听…';
        rec.onresult=e=>{let final='',interim='';for(let i=e.resultIndex;i<e.results.length;i++){const s=e.results[i][0].transcript;if(e.results[i].isFinal)final+=s;else interim+=s;} const target=el(targetId); if(final)target.value=(target.value+' '+final).trim(); button.title=interim;};
        rec.onerror=()=>{button.textContent=original;toast('语音识别没有成功，请再试一次。');};rec.onend=()=>button.textContent=original;rec.start();
      }
